127 template <typename T>
128 requires(!StringLike<T>)
129 Attribute<std::decay_t<T>> *add(const std::string &name, T &&value)
130 {
131 using ValueType = std::decay_t<T>;
132
133 auto attr = std::make_unique<Attribute<ValueType>>(name,
134 std::forward<T>(value));
135
136 auto *ptr = attr.get();
137
138 auto [it, inserted] = attributes_.try_emplace(name, std::move(attr));
139
140 if (!inserted)
141 throw std::runtime_error("Attribute already exists: " + name);
142
143 // keep track of insertion order
144 insertion_order_.push_back(name);
145
146 attribute_added.notify(*ptr);
147
148 return ptr;
149 }

296 template <typename T> T &value(const std::string &name)
